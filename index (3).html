function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet(); var data = JSON.parse(e.postData.contents); var totalKm = 0; var valorServico = 0; var valorTotal = 0; // Se for Desempeno ou tiver um valor forçado enviado pelo formulário: if (data.valorServicoForcado !== undefined) { totalKm = data.kmEfetuados || 0; valorServico = data.valorServicoForcado; valorTotal = data.valorTotalForcado; } // Se for Deslocação manual else if (data.tipoRegisto === "Deslocação") { totalKm = parseFloat(data.kmEfetuados); valorServico = totalKm < 25 ? data.precoLocalFixo : (totalKm * data.taxaAplicada); valorTotal = valorServico + parseFloat(data.portagens); } // Se for Serviço Normal (calcula distância pelo Google Maps API no Apps Script se aplicável) else { // Exemplo do cálculo normal do percurso totalKm = calcularDistanciaComGoogleMaps(data.base, data.origem, data.destino); // Usa a tua função existente se tiveres valorServico = totalKm < 25 ? data.precoLocalFixo : (totalKm * data.taxaAplicada); valorTotal = valorServico + parseFloat(data.portagens); } // Adiciona a linha à Folha de Cálculo sheet.appendRow([ data.dataHora, data.tipoRegisto, data.motorista, data.numTaxi, data.numProcesso, data.matricula, data.companhia, data.lotacao, data.origem, data.destino, totalKm, valorServico, data.portagens, valorTotal, data.observacoes ]); return ContentService.createTextOutput(JSON.stringify({ "result": "sucesso", "totalKm": totalKm, "valorServico": valorServico, "valorTotal": valorTotal })).setMimeType(ContentService.MimeType.JSON); }
